(e.parameter || {}) : {}; var action = params.action; if (action === "getStats") { return getDashboardStats(); } return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Invalid action"})).setMimeType(ContentService.MimeType.JSON); } function getDashboardStats() { try { var leaveAppUrl = "https://script.google.com/macros/s/AKfycbzl46S399DiMT_9D2xCLWNBJA440kZUPKKD5tpQjIPDDvLQWR7ZR_Kt5lMKnmusWJUG/exec"; var purchaseAppUrl = "https://script.google.com/macros/s/AKfycbzHCwnMeIS3KeLbo_cDvhyRKiO6RW9V2L_mCVIL2bW_7msJAA9beDQXCWTR27E94KoPgQ/exec"; // طلب إحصائيات الإجازات var leaveRes = UrlFetchApp.fetch(leaveAppUrl + "?action=get_leave_stats", { muteHttpExceptions: true }); var leaveData = {}; try { leaveData = JSON.parse(leaveRes.getContentText()); } catch(_) {} // طلب إحصائيات المشتريات var purchaseRes = UrlFetchApp.fetch(purchaseAppUrl + "?action=getPurchaseStats", { muteHttpExceptions: true }); var purchaseData = {}; try { purchaseData = JSON.parse(purchaseRes.getContentText()); } catch(_) {} return ContentService.createTextOutput(JSON.stringify({ status: "success", purchases: (purchaseData && purchaseData.data) ? purchaseData.data : {pending: 0, approved: 0, rejected: 0, total: 0}, leaves: (leaveData && leaveData.data) ? leaveData.data : {pending: 0, approved: 0, rejected: 0, total: 0} })).setMimeType(ContentService.MimeType.JSON); } catch (e) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: e.toString()})).setMimeType(ContentService.MimeType.JSON); } }
